// ============================================================ // BON BON - Puente entre la app y Google Sheets (version 4: con inventario) // Pega este codigo en tu Sheet: Extensiones -> Apps Script // // Cuando tocas REGISTRAR en la app: // 1) Se guarda el detalle del dia en la pestaña "Registro App" // 2) Se escribe la venta (amarillo) y la utilidad (verde) // en tu pestaña del año (2026) en el dia que corresponde // // Cada vez que el empleado toca ORDENAR en el Punto de Venta: // 3) Se resta del stock de la pestaña "Inventario" // 4) La orden queda anotada en la pestaña "Órdenes App" // // PRIMERA VEZ: ejecuta la funcion crearInventario para crear // la pestaña "Inventario" y llena la columna "Stock en refri". // ============================================================ // Cambia este PIN por uno tuyo. La app te lo pedira una sola vez. const PIN = "0123"; // Pestaña con un renglon por dia (se crea sola) const HOJA = "Registro App"; // Pestañas del inventario const HOJA_INV = "Inventario"; const HOJA_ORD = "Órdenes App"; // Utilidad aproximada para los dias que capturaste a mano (antes de la app) const FACTOR_UTILIDAD_MANUAL = 0.62; // Precio promedio para estimar BOCAS SATISFECHAS en dias capturados a mano const PRECIO_PROMEDIO = 70; // Hasta que columna llegan los dias del mes (AF = 32 columnas) const ULTIMA_COL = 32; const POSTRES = [ "Carlota", "Pay de Queso", "Flan Napolitano", "Tiramisu", "Besito de Nuez", "Cheesecake de Fresa", "Cheesecake de Oreo", "Pan de Elote" ]; const MESES_HOJA = ["ENE", "FEB", "MAR", "ABR", "MAY", "JUN", "JUL", "AGO", "SEP", "OCT", "NOV", "DIC"]; const DIAS_HOJA = ["DOM", "LUN", "MAR", "MIE", "JUE", "VIE", "SAB"]; // ------------------------------------------------------------ // WEB APP // ------------------------------------------------------------ // Abre la URL /exec en el navegador para comprobar que funciona function doGet() { return responder({ ok: true, mensaje: "Puente BON BON funcionando" }); } // La app llama aqui cuando tocas REGISTRAR function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const d = JSON.parse(e.postData.contents); if (String(d.pin) !== String(PIN)) return responder({ ok: false, error: "PIN" }); const ss = SpreadsheetApp.getActiveSpreadsheet(); // Ordenes del Punto de Venta -> inventario if (d.accion === "inventario") { const stock = procesarInventario_(ss, d.ops || []); return responder({ ok: true, stock: stock }); } // Fecha del dia registrado (a mediodia para evitar lios de zona horaria) const p = String(d.fecha).split("-").map(Number); const fecha = new Date(p[0], p[1] - 1, p[2], 12, 0, 0); const actualizado = guardarEnRegistro_(ss, fecha, d); let aviso = ""; try { escribirEnCalendario_(ss, fecha, r2(d.ventaTotal), r2(d.utilidad)); } catch (err) { aviso = "No se pudo escribir en la pestaña del año: " + err; } return responder({ ok: true, actualizado: actualizado, aviso: aviso }); } catch (err) { return responder({ ok: false, error: String((err && err.message) || err) }); } finally { lock.releaseLock(); } } // ------------------------------------------------------------ // INVENTARIO // Columna A = postre, columna B = stock en refri (lo llenas tu). // Si dejas vacio el stock de un postre, ese no se descuenta. // ------------------------------------------------------------ // EJECUTAR UNA VEZ desde el editor para crear la pestaña "Inventario" function crearInventario() { const ss = SpreadsheetApp.getActiveSpreadsheet(); hojaInventario_(ss); hojaOrdenes_(ss); } function productosInventario_() { return POSTRES.concat(["Fresas Chicas", "Fresas Grandes"]); } function hojaInventario_(ss) { let h = ss.getSheetByName(HOJA_INV); if (h) return h; h = ss.insertSheet(HOJA_INV); const prods = productosInventario_(); h.getRange(1, 1, 1, 3).setValues([["Postre", "Stock en refri", "Último movimiento"]]) .setFontWeight("bold").setBackground("#4F765F").setFontColor("#FFFFFF"); h.getRange(2, 1, prods.length, 1).setValues(prods.map(p => [p])).setFontWeight("bold"); h.getRange(2, 2, prods.length, 1).setBackground("#FFF4BD").setNumberFormat("0") .setHorizontalAlignment("center"); h.getRange(2, 3, prods.length, 1).setNumberFormat("dd/mm/yyyy hh:mm"); h.setFrozenRows(1); h.setColumnWidth(1, 190); h.setColumnWidth(2, 130); h.setColumnWidth(3, 150); // En rojo cuando quedan 2 o menos const regla = SpreadsheetApp.newConditionalFormatRule() .whenFormulaSatisfied("=AND(ISNUMBER($B2),$B2<=2)") .setBackground("#F4C7C3").setFontColor("#9C0006") .setRanges([h.getRange(2, 2, prods.length, 1)]) .build(); h.setConditionalFormatRules([regla]); return h; } function hojaOrdenes_(ss) { let h = ss.getSheetByName(HOJA_ORD); if (h) return h; h = ss.insertSheet(HOJA_ORD); h.getRange(1, 1, 1, 9).setValues([[ "ID", "Fecha", "Hora", "Orden #", "Detalle", "Total", "Estado", "Procesada el", "Piezas" ]]).setFontWeight("bold").setBackground("#C20034").setFontColor("#FFFFFF"); h.setFrozenRows(1); h.getRange("B2:B").setNumberFormat("dd/mm/yyyy"); h.getRange("F2:F").setNumberFormat("$#,##0.00"); h.getRange("H2:H").setNumberFormat("dd/mm/yyyy hh:mm"); h.setColumnWidth(5, 380); h.hideColumns(1); // ID interno h.hideColumns(9); // piezas (para poder regresarlas si se cancela) return h; } // Quita acentos y mayusculas para comparar nombres function clave_(t) { return String(t).trim().toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, ""); } function filasInventario_(inv) { const n = inv.getLastRow(); const mapa = {}; if (n < 2) return mapa; inv.getRange(2, 1, n - 1, 1).getValues().forEach((r, i) => { if (String(r[0]).trim()) mapa[clave_(r[0])] = i + 2; }); return mapa; } function leerStock_(inv) { const n = inv.getLastRow(); const stock = {}; if (n < 2) return stock; inv.getRange(2, 1, n - 1, 2).getValues().forEach(r => { if (String(r[0]).trim() && typeof r[1] === "number") stock[clave_(r[0])] = r[1]; }); return stock; } // signo = -1 para restar (venta), +1 para regresar (orden cancelada) function aplicarPiezas_(inv, filas, piezas, signo) { for (const nombre in piezas) { const fila = filas[clave_(nombre)]; if (!fila) continue; const celda = inv.getRange(fila, 2); const v = celda.getValue(); if (typeof v !== "number") continue; // vacio = no se lleva inventario celda.setValue(v + signo * (Number(piezas[nombre]) || 0)); inv.getRange(fila, 3).setValue(new Date()); } } // Cada orden trae un ID; si llega dos veces (por reintentos) se aplica una sola vez function procesarInventario_(ss, ops) { const inv = hojaInventario_(ss); const log = hojaOrdenes_(ss); const filas = filasInventario_(inv); const ids = {}; const n = log.getLastRow(); if (n >= 2) { log.getRange(2, 1, n - 1, 1).getValues().forEach((r, i) => { if (r[0]) ids[r[0]] = i + 2; }); } for (const op of ops) { if (!op || !op.id) continue; if (op.tipo === "orden") { if (ids[op.id]) continue; aplicarPiezas_(inv, filas, op.piezas || {}, -1); const p = String(op.fecha).split("-").map(Number); log.appendRow([ op.id, new Date(p[0], p[1] - 1, p[2], 12), op.hora || "", op.n || "", op.detalle || "", Number(op.total) || 0, "Aplicada", new Date(), JSON.stringify(op.piezas || {}) ]); ids[op.id] = log.getLastRow(); } else if (op.tipo === "cancelar") { const r = ids[op.id]; if (!r) { log.appendRow([op.id, "", "", "", "Cancelada antes de llegar", 0, "Cancelada", new Date(), "{}"]); ids[op.id] = log.getLastRow(); continue; } if (log.getRange(r, 7).getValue() === "Aplicada") { let piezas = {}; try { piezas = JSON.parse(log.getRange(r, 9).getValue() || "{}"); } catch (e) {} aplicarPiezas_(inv, filas, piezas, +1); log.getRange(r, 7).setValue("Cancelada"); } } } return leerStock_(inv); } // ------------------------------------------------------------ // PESTAÑA "Registro App" (un renglon por dia) // ------------------------------------------------------------ // Columnas de la pestaña "Registro App" (en este orden) function encabezadosRegistro_() { return [ "Fecha", "Venta total", "Costo postres", "Utilidad", "Postres vendidos", "Empleado 18%", "Renta + Luz + Gas", "Profit libre", "Profit si vendiste tú" ].concat(POSTRES).concat(["Fresas Chicas", "Fresas Grandes", "Registrado el"]); } function guardarEnRegistro_(ss, fecha, d) { const hoja = ss.getSheetByName(HOJA) || crearHojaRegistro_(ss); const columnas = asegurarColumnas_(hoja); // nombre -> numero de columna const valores = { "Fecha": fecha, "Venta total": r2(d.ventaTotal), "Costo postres": r2(d.costoPostres), "Utilidad": r2(d.utilidad), "Postres vendidos": Number(d.totalPostres) || 0, "Empleado 18%": r2(d.comisionEmpleado), "Renta + Luz + Gas": r2(d.gastosFijos), "Profit libre": r2(d.profitLibre), "Profit si vendiste tú": r2(d.profitSinEmpleado), "Fresas Chicas": Number(d.fresasChicas) || 0, "Fresas Grandes": Number(d.fresasGrandes) || 0, "Registrado el": new Date() }; POSTRES.forEach((p, i) => { valores[p] = Number((d.vendidos || [])[i]) || 0; }); // Si ese dia ya estaba registrado, se reemplaza (no se duplica) const tz = ss.getSpreadsheetTimeZone(); const clave = Utilities.formatDate(fecha, tz, "yyyy-MM-dd"); const ultima = hoja.getLastRow(); let destino = 0; if (ultima >= 2) { const fechas = hoja.getRange(2, 1, ultima - 1, 1).getValues(); for (let i = 0; i < fechas.length; i++) { const f = fechas[i][0]; if (f instanceof Date && Utilities.formatDate(f, tz, "yyyy-MM-dd") === clave) { destino = i + 2; break; } } } const actualizado = destino > 0; if (!actualizado) destino = ultima + 1; // Se escribe por nombre de columna; lo demas del renglon se respeta const ancho = hoja.getLastColumn(); const fila = actualizado ? hoja.getRange(destino, 1, 1, ancho).getValues()[0] : new Array(ancho).fill(""); for (const nombre in valores) { if (columnas[nombre]) fila[columnas[nombre] - 1] = valores[nombre]; } hoja.getRange(destino, 1, 1, ancho).setValues([fila]); if (hoja.getLastRow() > 2) { hoja.getRange(2, 1, hoja.getLastRow() - 1, hoja.getLastColumn()).sort(1); } return actualizado; } // Agrega las columnas que falten (por ejemplo las de fresas) antes de "Registrado el" function asegurarColumnas_(hoja) { const leer = () => { const ancho = Math.max(hoja.getLastColumn(), 1); const enc = hoja.getRange(1, 1, 1, ancho).getDisplayValues()[0]; const mapa = {}; enc.forEach((t, i) => { if (String(t).trim()) mapa[String(t).trim()] = i + 1; }); return mapa; }; let mapa = leer(); for (const nombre of encabezadosRegistro_()) { if (mapa[nombre]) continue; let col; if (mapa["Registrado el"]) { col = mapa["Registrado el"]; hoja.insertColumnBefore(col); } else { col = hoja.getLastColumn() + 1; } hoja.getRange(1, col).setValue(nombre) .setFontWeight("bold").setBackground("#4F765F").setFontColor("#FFFFFF"); hoja.getRange(2, col, hoja.getMaxRows() - 1, 1).setNumberFormat("0"); mapa = leer(); } return mapa; } function crearHojaRegistro_(ss) { const hoja = ss.insertSheet(HOJA); const encabezados = encabezadosRegistro_(); hoja.getRange(1, 1, 1, encabezados.length) .setValues([encabezados]) .setFontWeight("bold") .setBackground("#4F765F") .setFontColor("#FFFFFF"); hoja.setFrozenRows(1); const ultimaCol = encabezados.length; hoja.getRange("A2:A").setNumberFormat("dd/mm/yyyy"); hoja.getRange("B2:D").setNumberFormat("$#,##0.00"); hoja.getRange("F2:I").setNumberFormat("$#,##0.00"); hoja.getRange(2, ultimaCol, hoja.getMaxRows() - 1, 1).setNumberFormat("dd/mm/yyyy hh:mm"); return hoja; } // ------------------------------------------------------------ // PESTAÑA DEL AÑO (2026): bloques de 4 renglones por mes // renglon del mes -> "NOV" | TOTAL MES | =SUMA // renglon de dias -> DOM 1, MAR 3, ... // renglon amarillo -> venta del dia // renglon verde -> utilidad del dia // ------------------------------------------------------------ function escribirEnCalendario_(ss, fecha, venta, utilidad) { const hoja = ss.getSheetByName(String(fecha.getFullYear())) || crearHojaAnio_(ss, fecha.getFullYear()); const filaMes = buscarMes_(hoja, MESES_HOJA[fecha.getMonth()]) || crearBloqueMes_(hoja, MESES_HOJA[fecha.getMonth()]); const filaDias = filaMes + 1, filaVenta = filaMes + 2, filaUtil = filaMes + 3; const etiquetas = hoja.getRange(filaDias, 1, 1, ULTIMA_COL).getDisplayValues()[0]; const dia = fecha.getDate(); let col = 0, insertarEn = 0, ultimaUsada = 0; for (let c = 0; c < etiquetas.length; c++) { const n = numeroDeDia_(etiquetas[c]); if (!n) continue; ultimaUsada = c + 1; if (n === dia) { col = c + 1; break; } if (n > dia && !insertarEn) insertarEn = c + 1; } if (!col) { if (insertarEn) { // Dia que falta en medio: se recorren a la derecha solo los 3 renglones del mes hoja.getRange(filaDias, insertarEn, 3, 1) .insertCells(SpreadsheetApp.Dimension.COLUMNS); col = insertarEn; } else { col = ultimaUsada + 1; } // Copiar colores y formato de una columna vecina const vecina = col > 1 ? col - 1 : col + 1; hoja.getRange(filaDias, vecina, 3, 1) .copyFormatToRange(hoja, col, col, filaDias, filaUtil); } const etiqueta = DIAS_HOJA[fecha.getDay()] + " " + dia; hoja.getRange(filaDias, col).setNumberFormat("@").setValue(etiqueta); hoja.getRange(filaVenta, col).setNumberFormat("$#,##0.00").setValue(venta); hoja.getRange(filaUtil, col).setNumberFormat("$#,##0.00").setValue(utilidad); } function numeroDeDia_(texto) { const m = String(texto).trim().match(/(\d{1,2})\s*$/); return m ? Number(m[1]) : 0; } function buscarMes_(hoja, mes) { const n = Math.max(hoja.getLastRow(), 1); const colA = hoja.getRange(1, 1, n, 1).getDisplayValues(); for (let i = 0; i < colA.length; i++) { if (String(colA[i][0]).trim().toUpperCase() === mes) return i + 1; } return 0; } // Los meses van en los renglones 2, 6, 10, 14... (cada 4) function crearBloqueMes_(hoja, mes) { let r = hoja.getLastRow() + 1; while (r % 4 !== 2) r++; hoja.getRange(r, 1).setValue(mes).setFontWeight("bold"); ponerTotalMes_(hoja, r); return r; } function ponerTotalMes_(hoja, filaMes) { const filaVenta = filaMes + 2; hoja.getRange(filaMes, 2).setValue("TOTAL MES"); hoja.getRange(filaMes, 3) .setFormula("=SUM(A" + filaVenta + ":" + letra_(ULTIMA_COL) + filaVenta + ")") .setNumberFormat("$#,##0.00"); } function crearHojaAnio_(ss, anio) { const hoja = ss.insertSheet(String(anio)); ponerTotalesArriba_(hoja); return hoja; } // Totales del renglon 1, solo con funciones normales de Sheets function ponerTotalesArriba_(hoja) { const rango = "A2:" + letra_(ULTIMA_COL) + "400"; hoja.getRange("A1").setValue("TOTAL VENDIDO"); hoja.getRange("C1").setFormula( "=ARRAYFORMULA(SUM(IF(MOD(ROW(" + rango + "),4)=0,IF(ISNUMBER(" + rango + ")," + rango + ",0),0)))" ).setNumberFormat("$#,##0.00"); hoja.getRange("E1").setValue("UTILIDAD"); hoja.getRange("F1").setFormula( "=ARRAYFORMULA(SUM(IF(MOD(ROW(" + rango + "),4)=1,IF(ISNUMBER(" + rango + ")," + rango + ",0),0)))" ).setNumberFormat("$#,##0.00"); hoja.getRange("H1").setValue("BOCAS SATISFECHAS"); // Dias a mano: venta / precio promedio. Dias con la app: postres reales. const anio = Number(hoja.getName()) || new Date().getFullYear(); const R = "'" + HOJA + "'!"; const enAnio = R + "A:A,\">=\"&DATE(" + anio + ",1,1)," + R + "A:A,\"<\"&DATE(" + (anio + 1) + ",1,1)"; hoja.getRange("J1").setFormula( "=IFERROR(ROUND((C1-SUMIFS(" + R + "B:B," + enAnio + "))/" + PRECIO_PROMEDIO + "+SUMIFS(" + R + "E:E," + enAnio + "),0),ROUND(C1/" + PRECIO_PROMEDIO + ",0))" ).setNumberFormat("0"); hoja.getRange("L1").setValue("LIBRES"); hoja.getRange("N1").setFormula("=COUNTIF(" + rango + ",\"libre\")") .setNumberFormat("0"); } // ------------------------------------------------------------ // EJECUTAR UNA SOLA VEZ desde el editor: repararFormulas // - Arregla los totales de arriba (los #NAME?) // - Pone TOTAL MES en cada mes // - Cambia las utilidades con #VALUE! por una formula que no falla // - Crea la pestaña "Registro App" si no existe // ------------------------------------------------------------ function repararFormulas() { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss.getSheetByName(HOJA)) crearHojaRegistro_(ss); asegurarColumnas_(ss.getSheetByName(HOJA)); const hoja = ss.getSheetByName("2026"); if (!hoja) throw new Error("No encontre la pestaña 2026"); ponerTotalesArriba_(hoja); const n = hoja.getLastRow(); const colA = hoja.getRange(1, 1, n, 1).getDisplayValues(); for (let i = 0; i < n; i++) { const mes = String(colA[i][0]).trim().toUpperCase(); if (MESES_HOJA.indexOf(mes) < 0) continue; const filaMes = i + 1; const filaDias = filaMes + 1, filaVenta = filaMes + 2, filaUtil = filaMes + 3; ponerTotalMes_(hoja, filaMes); const etiquetas = hoja.getRange(filaDias, 1, 1, ULTIMA_COL).getDisplayValues()[0]; const util = hoja.getRange(filaUtil, 1, 1, ULTIMA_COL); const formulas = util.getFormulas()[0]; const valores = util.getValues()[0]; for (let c = 0; c < ULTIMA_COL; c++) { const celda = hoja.getRange(filaUtil, c + 1); if (!String(etiquetas[c]).trim()) { // Columna sin dia: quitar los $0.00 sobrantes (se quedan los colores) if (formulas[c] || valores[c] !== "") celda.clearContent(); continue; } // Los numeros fijos (los que escribe la app) se respetan const esNumeroFijo = !formulas[c] && typeof valores[c] === "number"; if (esNumeroFijo) continue; const ref = letra_(c + 1) + filaVenta; celda.setFormula("=IF(ISNUMBER(" + ref + ")," + ref + "*" + FACTOR_UTILIDAD_MANUAL + ",\"\")") .setNumberFormat("$#,##0.00"); } } } // ------------------------------------------------------------ // UTILIDADES // ------------------------------------------------------------ function letra_(col) { let s = ""; while (col > 0) { const m = (col - 1) % 26; s = String.fromCharCode(65 + m) + s; col = Math.floor((col - 1) / 26); } return s; } function r2(n) { return Math.round((Number(n) || 0) * 100) / 100; } function responder(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
